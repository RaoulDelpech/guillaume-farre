## INTÉGRATION TECHNIQUE (aperçu)

### Architecture recommandée

```
Client paye via Stripe
    ↓
Stripe webhook → /api/stripe-webhook
    ↓
Créer commande via ThePrintSpace API
    ↓
ThePrintSpace produit + expédie
    ↓
Webhook ThePrintSpace → /api/print-webhook
    ↓
Mettre à jour commande (tracking)
    ↓
Email client avec tracking
```

### Exemple code

```typescript
// /app/api/stripe-webhook/route.ts
import Stripe from 'stripe';

export async function POST(req: Request) {
  const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!);
  const sig = req.headers.get('stripe-signature')!;

  const event = stripe.webhooks.constructEvent(
    await req.text(),
    sig,
    process.env.STRIPE_WEBHOOK_SECRET!
  );

  if (event.type === 'checkout.session.completed') {
    const session = event.data.object;

    // Créer commande ThePrintSpace
    const printOrder = await createPrintOrder({
      productId: session.metadata.productId,
      size: session.metadata.size,
      finish: session.metadata.finish,
      shippingAddress: session.shipping_details,
    });

    // Sauvegarder tracking
    await saveOrderTracking({
      stripeSessionId: session.id,
      printOrderId: printOrder.id,
      status: 'processing',
    });
  }

  return new Response(JSON.stringify({ received: true }));
}

// /lib/theprintspace-api.ts
async function createPrintOrder(params) {
  const response = await fetch('https://api.theprintspace.com/v1/orders', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${process.env.THEPRINTSPACE_API_KEY}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      product: params.productId,
      size: params.size,
      finish: params.finish,
      shipping: params.shippingAddress,
      // ...
    }),
  });

  return response.json();
}
```
